// The digits of a duration part: '_' only between two digits, removed.
bool DurationDigits(std::string_view s, std::string& out) {
    std::string b;
    for (size_t i = 0; i < s.size(); ++i) {
        if (s[i] != '_') {
            b.push_back(s[i]);
        } else if (i == 0 || i + 1 >= s.size() || !IsDecDigit(s[i - 1]) || !IsDecDigit(s[i + 1])) {
            return false;
        }
    }
    if (!AllDigits(b)) return false;
    out = b;
    return true;
}
