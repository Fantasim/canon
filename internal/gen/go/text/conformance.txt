{{define "conformance"}}// canonCatch runs f and returns its result, or the code of the *rt.EvalError it
// panicked with. Any other panic is re-raised.
func canonCatch[T any](f func() T) (got T, code string) {
	defer func() {
		if r := recover(); r != nil {
			e, ok := r.(*{{.RT}}.EvalError)
			if !ok {
				panic(r)
			}
			code = e.Code
		}
	}()
	return f(), ""
}
