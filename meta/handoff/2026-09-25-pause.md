### load.py

```python
# Orchestrator-authored spec text (DECISIONS 207). Run from the repository root.
p='spec/ERRORS.md'; s=open(p).read()
def rep(old,new):
    global s
    assert old in s, 'not found: '+old[:70]
    s=s.replace(old,new,1)
rep('| E7004 | - | path:Path, cause:Text | `cannot read {path}: {cause}` |','| E7004 | - | path:Path, cause:Kind | `cannot read {path}: {cause}` |')
rep('| E7005 | - | pattern:Text, cause:Text | `invalid glob {pattern}: {cause}` |','| E7005 | - | pattern:Text, cause:Kind | `invalid glob {pattern}: {cause}` |')
rep('| W7115 | - | path:Path | `symbolic link {path} points outside the roots; skipped` |',
'''| W7115 | outsideRoots | path:Path | `symbolic link {path} points outside the roots; skipped` |
| W7115 | dangling | path:Path | `symbolic link {path} points to nothing; skipped` |
| W7115 | looping | path:Path | `symbolic link {path} is part of a loop; skipped` |
| W7115 | statFailed | path:Path | `symbolic link {path} resolves but cannot be read; skipped` |''')
rep('| W7115 | warning | load | WIRE.md §6.5 | a symbolic link pointing outside the roots is skipped |',
    '| W7115 | warning | load | WIRE.md §6.5 | a symbolic link skipped: outside the roots, dangling, looping or unreadable |')
# Kinds (§1.4), inserted in alphabetical order by the builder? No: inserted here, sorted.
kinds=[('GlobBrace','an unclosed, empty or nested brace','`E7005`'),
       ('GlobBracket','an unclosed bracket','`E7005`'),
       ('GlobDoubleStar','a double star that is not a whole segment','`E7005`'),
       ('ReadMissing','no such file or directory','`E7004`'),
       ('ReadNotDir','not a directory','`E7004`'),
       ('ReadPermission','permission denied','`E7004`'),
       ('ReadTooLarge','too large','`E7004`'),
       ('ReadUnreadable','unreadable','`E7004`')]
start=s.index('| Kind | Word | Used by |'); end=s.index('\n\nA constructor is given only a Kind', start)
block=s[start:end].split('\n'); head=block[:2]; rows=block[2:]
for k,w,u in kinds: rows.append(f'| `{k}` | {w} | {u} |')
rows.sort(key=lambda r: r.split('`')[1])
s=s[:start]+'\n'.join(head+rows)+s[end:]
open(p,'w').write(s)
rep('| E7001 | empty | path:Text | `path {path}: empty segment` |','''| E7001 | empty | path:Text | `path {path}: empty segment` |
| E7001 | dot | path:Text, seg:Text | `path {path}: "{seg}" is not a path segment` |''')
open(p,'w').write(s)
print('ok: update the count sentence to what diaggen reports')
```

### gate.sh (the orchestrator's make check wrapper)

```sh
#!/bin/bash
# ORCHESTRATOR ONLY. usage: gate.sh <dir> <log> — make check under the cap, fresh lint cache
cd "$1" || exit 2
LC=$(mktemp -d -p /var/tmp); TD=$(mktemp -d -p /var/tmp)
systemd-run --user --scope -q -p MemoryMax=3G env GOTOOLCHAIN=local GOLANGCI_LINT_CACHE=$LC TMPDIR=$TD make check > "$2" 2>&1
st=$?
rm -rf "$LC" "$TD"
grep -E -- '--- FAIL|^FAIL|make: \*\*\*|^NEW|NEW  |not formatted' "$2" | head -30
if grep -q "not measured" "$2"; then echo "AUDIT NOT FULLY MEASURED"; st=1; fi
tail -2 "$2"
echo "gate($1) exit=$st"
exit $st
```
