const chainSrc = `package d

/// What a quest counts.
enum Goal { kill, collect }

/// What a quest aims at, by goal.
type Target(g: Goal) = match g {
  kill => String(1..)
  collect => Int(1..)
}

/// A part.
record Sub(g: Goal) {
  /// Its kind.
  kind: Goal
  /// Its aim.
  aim: Target(g)?
}

/// A quest.
record Quest {
  /// What it counts.
  goal: Goal
  /// Its part.
  sub: Sub(goal)
  /// Extra, by the part's kind.
  extra: Target(sub.kind)?
}

/// The quests.
let quests: table Quest = {
  slay { goal: kill, sub: { kind: kill }, extra: "x" }
  other { goal: kill, sub: { kind: kill } }
}
`

func chainFS() mapFS {
	return mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(chainSrc)}
}

func pset(path string, v edit.Lit) edit.Operation {
	return edit.Operation{Kind: edit.OpSet, Path: path, Value: v}
}

func TestProbeChain(t *testing.T) {
	s := open(t, chainFS(), nil, "", "d")
	_ = s
	collect := edit.Member("collect")
	undoTwice(t, "chain", chainFS(), []edit.Operation{
		pset("quests.slay.goal", collect), pset("quests.slay.extra", edit.Str("y")), pset("quests.slay.sub.kind", collect),
	})
}

func TestProbeRemoveBetween(t *testing.T) {
	collect := edit.Member("collect")
	undoTwice(t, "remove between", questFS(), []edit.Operation{
		pset("quests.slay.goal", collect), {Kind: edit.OpRemove, Path: "quests.gather"}, pset("quests.slay.target", edit.Int(25)),
	})
}

func dumpFindings(t *testing.T, fs mapFS) {
	s := open(t, fs, nil, "", "d")
	for _, f := range s.a.Bag("d").Findings() {
		t.Logf("%v", f)
	}
}

func TestProbeChainFindings(t *testing.T) { dumpFindings(t, chainFS()) }

const prSrc = `package d

/// What a quest counts.
enum Goal { kill, collect }

/// What a quest aims at, by goal.
type Target(g: Goal) = match g {
  kill => String(1..)
  collect => Int(1..)
}

/// A part.
record Sub(g: Goal) {
  /// Its kind.
  kind: Goal
  /// Its aim.
  aim: Target(kind)?
  /// Its other aim, by the quest's goal.
  far: Target(g)?
}

/// A quest.
record Quest {
  /// What it counts.
  goal: Goal
  /// Its part.
  sub: Sub(goal)
}

/// The quests.
let quests: table Quest = {
  slay { goal: kill, sub: { kind: kill } }
}
`

func prFS() mapFS { return mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(prSrc)} }

func TestProbePR(t *testing.T) {
	dumpFindings(t, prFS())
	collect := edit.Member("collect")
	ops := []edit.Operation{pset("quests.slay.goal", collect), pset("quests.slay.sub.aim", edit.Str("y")), pset("quests.slay.sub.kind", collect)}
	e, ok := applied(t, "pr", prFS(), ops)
	if ok {
		for _, u := range e.plan.Undo {
			t.Logf("undo %+v", u)
		}
		t.Logf("dropped %+v", e.plan.Dropped)
	}
	undoTwice(t, "pr", prFS(), ops)
}

func logPlan(t *testing.T, name string, fs mapFS, ops []edit.Operation) {
	e, ok := applied(t, name, fs, ops)
	if !ok {
		return
	}
	for _, u := range e.plan.Undo {
		t.Logf("%s undo %v %s %v %v", name, u.Kind, u.Path, u.Value, u.Key)
	}
	t.Logf("%s dropped %+v", name, e.plan.Dropped)
}

const twoDrvSrc = `package d

/// What a quest counts.
enum Goal { kill, collect }

/// By flag, for kill.
type KF(f: Bool) = match f {
  false => String(1..)
  true => Int(1..9)
}

/// By flag, for collect.
type CF(f: Bool) = match f {
  false => Int(10..)
  true => Bool
}

/// By goal and flag.
type T2(g: Goal, f: Bool) = match g {
  kill => KF(f)
  collect => CF(f)
}

/// A quest.
record Quest {
  /// Goal.
  goal: Goal
  /// Flag.
  flag: Bool = false
  /// Aim.
  aim: T2(goal, flag)?
}

/// The quests.
let quests: table Quest = {
  slay { goal: kill, aim: "wolf" }
  other { goal: collect, aim: 12 }
}
`

func TestProbeMore(t *testing.T) {
	collect, kill := edit.Member("collect"), edit.Member("kill")
	q := questFS()
	cases := []struct {
		name string
		fs   mapFS
		ops  []edit.Operation
	}{
		{"dep, driver, dep", q, []edit.Operation{pset("quests.slay.target", edit.Str("boar")), pset("quests.slay.goal", collect), pset("quests.slay.target", edit.Int(25))}},
		{"driver back", q, []edit.Operation{pset("quests.slay.goal", collect), pset("quests.slay.target", edit.Int(25)), pset("quests.slay.goal", kill)}},
		{"driver back no dep", q, []edit.Operation{pset("quests.slay.goal", collect), pset("quests.slay.goal", kill)}},
		{"driver back reset", q, []edit.Operation{pset("quests.slay.goal", collect), {Kind: edit.OpReset, Path: "quests.slay.target"}, pset("quests.slay.goal", kill)}},
		{"list remove then set", q, []edit.Operation{pset("quests.slay.items", edit.Bool(true)), {Kind: edit.OpRemove, Path: "quests.slay.rewards[0]"}, pset("quests.slay.rewards[0]", edit.Str("gem"))}},
		{"list insert", q, []edit.Operation{pset("quests.slay.items", edit.Bool(true)), pset("quests.slay.rewards", edit.Source(`[]`)), {Kind: edit.OpInsert, Path: "quests.slay.rewards", Index: 0, Value: edit.Str("gem")}}},
		{"list add", q, []edit.Operation{pset("quests.slay.items", edit.Bool(true)), pset("quests.slay.rewards", edit.Source(`["a"]`)), {Kind: edit.OpAdd, Path: "quests.slay.rewards", Value: edit.Str("gem")}}},
		{"addentry then set it, then remove other", q, []edit.Operation{
			{Kind: edit.OpAddEntry, Path: "quests", Key: edit.Key("hunt"), Value: edit.Source(`{ goal: kill, target: "x", act: wait { turns: 1 } }`)},
			pset("quests.hunt.goal", collect), pset("quests.hunt.target", edit.Int(4)),
		}},
		{"remove entry then re-add", q, []edit.Operation{
			pset("quests.slay.goal", collect), {Kind: edit.OpRemove, Path: "quests.slay"},
			{Kind: edit.OpAddEntry, Path: "quests", Key: edit.Key("slay"), Value: edit.Source(`{ goal: kill, target: "y", act: wait { turns: 1 } }`)},
		}},
		{"setcase twice", q, []edit.Operation{
			{Kind: edit.OpSetCase, Path: "quests.slay.act", Case: "hunt", Value: edit.Obj{"goal": collect}},
			pset("quests.slay.act.target", edit.Int(5)),
			pset("quests.slay.act.goal", kill),
		}},
		{"case driver then whole act", q, []edit.Operation{
			pset("quests.gather.act.goal", collect), pset("quests.gather.act.target", edit.Int(5)),
			pset("quests.gather.act", edit.Source(`hunt { goal: kill, target: "deer" }`)),
		}},
		{"two drivers a", mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(twoDrvSrc)}, []edit.Operation{
			pset("quests.slay.goal", collect), pset("quests.slay.flag", edit.Bool(true)), pset("quests.slay.aim", edit.Bool(true)),
		}},
		{"two drivers b", mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(twoDrvSrc)}, []edit.Operation{
			pset("quests.slay.flag", edit.Bool(true)), pset("quests.slay.aim", edit.Int(3)), pset("quests.slay.goal", collect), pset("quests.slay.aim", edit.Bool(false)),
		}},
		{"two drivers c", mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(twoDrvSrc)}, []edit.Operation{
			pset("quests.slay.goal", collect), pset("quests.slay.aim", edit.Int(50)), pset("quests.slay.flag", edit.Bool(true)),
		}},
		{"two drivers d", mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(twoDrvSrc)}, []edit.Operation{
			pset("quests.slay.flag", edit.Bool(true)), pset("quests.slay.aim", edit.Int(3)), pset("quests.slay.flag", edit.Bool(false)), pset("quests.slay.goal", collect), pset("quests.slay.aim", edit.Int(11)),
		}},
	}
	for _, c := range cases {
		logPlan(t, c.name, c.fs, c.ops)
		undoTwice(t, c.name, c.fs, c.ops)
	}
}

func TestProbeJSON(t *testing.T) {
	m := func(s string) edit.Lit { return edit.Member(s) }
	set := pset
	for _, c := range []struct {
		name string
		fsys func() mapFS
		ops  []edit.Operation
	}{
		{"tag driver then dep", depFS, []edit.Operation{set("root.tag.goal", m("kill")), set("root.tag.pick", edit.Key("wolf"))}},
		{"tag dep then driver", depFS, []edit.Operation{set("root.tag.pick", m("green")), set("root.tag.goal", m("kill"))}},
		{"tag driver, dep, driver", depFS, []edit.Operation{set("root.tag.goal", m("kill")), set("root.tag.pick", edit.Key("wolf")), set("root.tag.goal", m("paint"))}},
		{"tag driver, dep, driver back to kill", depFS, []edit.Operation{set("root.tag.goal", m("kill")), set("root.tag.pick", edit.Key("wolf")), set("root.tag.goal", m("count")), set("root.tag.goal", m("kill"))}},
		{"ref driver then dep", depFS, []edit.Operation{set("objectives.o1.def", edit.Key("art")), set("objectives.o1.target", m("green"))}},
		{"ref driver then dep then note", depFS, []edit.Operation{set("objectives.o1.def", edit.Key("art")), set("objectives.o1.note", edit.Str("x")), set("objectives.o1.target", m("green"))}},
		{"held: visit driver then dep", undoFS, []edit.Operation{set("root.tag.goal", m("paint")), set("root.tag.pick", m("green"))}},
		{"held: dep then driver", undoFS, []edit.Operation{set("root.tag.pick", edit.FromJSON(`"q"`)), set("root.tag.goal", m("paint"))}},
		{"held: driver twice", undoFS, []edit.Operation{set("root.tag.goal", m("paint")), set("root.tag.goal", m("count"))}},
		{"held: driver, other, driver back", undoFS, []edit.Operation{set("root.tag.goal", m("paint")), set("objectives.o1.note", edit.Str("x")), set("root.tag.goal", m("visit"))}},
		{"held objs: def then target", undoFS, []edit.Operation{set("objectives.o2.def", edit.Key("art")), set("objectives.o2.target", m("red"))}},
		{"held objs: target then def", undoFS, []edit.Operation{set("objectives.o2.target", edit.FromJSON(`"elsewhere"`)), set("objectives.o2.def", edit.Key("art"))}},
		{"list elem: def then alts add", depFS, []edit.Operation{set("objectives.o1.def", edit.Key("art")), {Kind: edit.OpAdd, Path: "objectives.o1.alts", Value: m("red")}}},
		{"addentry dep map then driver", depFS, []edit.Operation{{Kind: edit.OpAddEntry, Path: "root.tag.byPick", Key: m("green"), Value: edit.Int(1)}, set("root.tag.goal", m("kill"))}},
		{"driver then addentry dep map", depFS, []edit.Operation{set("root.tag.goal", m("kill")), {Kind: edit.OpAddEntry, Path: "root.tag.byPick", Key: edit.Key("wolf"), Value: edit.Int(1)}}},
		{"objs list: driver, remove other, dep", depFS, []edit.Operation{
			{Kind: edit.OpAdd, Path: "root.objs", Value: edit.Source("{ def: hunt, target: wolf }")},
			set("root.objs[0].def", edit.Key("hunt")), {Kind: edit.OpRemove, Path: "root.objs[1]"}, set("root.objs[0].target", edit.Key("bear")),
		}},
		{"objs list: remove first, driver, dep", depFS, []edit.Operation{
			{Kind: edit.OpInsert, Path: "root.objs", Index: 0, Value: edit.Source("{ def: hunt, target: wolf }")},
			set("root.objs[1].def", edit.Key("hunt")), set("root.objs[1].target", edit.Key("bear")),
		}},
	} {

	}

const layerQuestSrc = `package d

/// What a quest counts.
enum Goal { kill, collect }

/// What a quest aims at, by goal.
type Target(g: Goal) = match g {
  kill => String(1..)
  collect => Int(1..)
}

/// A quest.
record Quest {
  /// What it counts.
  goal: Goal
  /// What it aims at.
  target: Target(goal)?
  /// A note.
  note: String = ""
}

/// One quest.
let q: Quest = { goal: kill, target: "wolf" }

/// Another quest.
let r: Quest = { goal: kill, target: "wolf" }
`

const layerQuestDev = `package d
layer dev

amend r {
  goal: collect
  target: 5
}
`

func layerUndoTwice(t *testing.T, name string, fsys mapFS, ops []edit.Operation) {
	t.Helper()
	layers := []string{"dev"}
	ap := func(n string, fs mapFS, ops []edit.Operation) (*edit.Plan, mapFS, bool) {
		s := open(t, fs, layers, "dev", "d")
		plan, err := edit.Apply(context.Background(), s.env, s.snap, edit.Request{Ops: ops})
		if err != nil {
			t.Errorf("%s: %+v: %v", n, ops, err)
			return nil, nil, false
		}
		for _, u := range plan.Undo {
			t.Logf("%s undo %v %s %v", n, u.Kind, u.Path, u.Value)
		}
		return plan, written(fs, plan), true
	}
	p1, f1, ok := ap(name, fsys, ops)
	if !ok {
		return
	}
	p2, f2, ok := ap(name+" undo", f1, p1.Undo)
	if !ok {
		return
	}
	cmp := func(n string, want, got mapFS) {
		w, g := values(open(t, want, layers, "dev", "d"), false), values(open(t, got, layers, "dev", "d"), false)
		for k, v := range w {
			if g[k] != v {
				t.Errorf("%s: %s = %s, want %s", n, k, g[k], v)
			}
		}
	}
	cmp(name+" undo", fsys, f2)
	_, f3, ok := ap(name+" redo", f2, p2.Undo)
	if ok {
		cmp(name+" redo", f1, f3)
	}
}

func TestProbeLayer(t *testing.T) {
	fs := func() mapFS {
		return mapFS{"law/project.canon": file(projectCanon), "law/d/d.canon": file(layerQuestSrc), "law/d/dev.layer.canon": file(layerQuestDev)}
	}
	collect, kill := edit.Member("collect"), edit.Member("kill")
	layerUndoTwice(t, "layer q driver dep", fs(), []edit.Operation{pset("q.goal", collect), pset("q.target", edit.Int(25))})
	layerUndoTwice(t, "layer q dep driver", fs(), []edit.Operation{pset("q.target", edit.Str("boar")), pset("q.goal", collect)})
	layerUndoTwice(t, "layer r driver dep", fs(), []edit.Operation{pset("r.goal", kill), pset("r.target", edit.Str("x"))})
	layerUndoTwice(t, "layer r reset driver", fs(), []edit.Operation{{Kind: edit.OpReset, Path: "r.goal"}, pset("r.target", edit.Str("x"))})
	layerUndoTwice(t, "layer r reset both", fs(), []edit.Operation{{Kind: edit.OpReset, Path: "r.goal"}, {Kind: edit.OpReset, Path: "r.target"}})
}
